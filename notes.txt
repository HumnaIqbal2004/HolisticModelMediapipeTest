Pose Landmarker code 
import math
import cv2
import numpy as np
from time import time
import mediapipe as mp
import matplotlib.pyplot as pit

BaseOptions = mp.tasks.BaseOptions
PoseLandmarker = mp.tasks.vision.PoseLandmarker
PoseLandmarkerOptions = mp.tasks.vision.PoseLandmarkerOptions
VisionRunningMode = mp.tasks.vision.RunningMode

options = PoseLandmarkerOptions (
    base_options = BaseOptions(model_asset_path = "pose_landmarker_full.task"),
    running_mode = VisionRunningMode.VIDEO,
    num_poses = 1
)

# Pairs of joint indices that connect to form bones
POSE_CONNECTIONS = [
    # Shoulders and arms
    (11, 12), (11, 13), (13, 15), (15,21), (15, 17), (15, 19), (19, 17), # Left arm and hands
    (12, 14), (14, 16), (16, 22), (16, 18), (16, 20), (18, 20),          # Right arm and hands
    # Torso
    (11, 23), (12, 24), (23, 24),
    # Legs
    (23, 25), (25, 27),            # Left leg
    (24, 26), (26, 28),            # Right leg
]



cap = cv2.VideoCapture(0)

with PoseLandmarker.create_from_options(options) as landmarker:
    frame_timestamp_ms = 0


    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            print("Couldn't get webcam")
            break

        frame = cv2.flip(frame, 1)

        height, width, _ = frame.shape

        rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) #opencv uses brg 

        mp_image = mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb_frame) #convert image into a mediapipe object

        frame_timestamp_ms += 33

        result = landmarker.detect_for_video(mp_image, frame_timestamp_ms)
        

        if result.pose_landmarks:
            landmarks = result.pose_landmarks[0]

            for start_idx, end_idx in POSE_CONNECTIONS: #so it walks through the pair in the pose_connects array
                pt1 = landmarks[start_idx] 
                pt2 = landmarks[end_idx]
                x1, y1 = int(pt1.x * width), int(pt1.y * height)
                x2, y2 = int(pt2.x * width), int(pt2.y * height)
                cv2.line(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)  # Green lines
            
            for lm in landmarks:
                cx, cy = int(lm.x * width), int(lm.y * height)
                cv2.circle(frame, (cx, cy), 5, (0, 0, 255), -1)

        #Add a crash prevention

        cv2.imshow("Webcam Pose", frame)

        if cv2.waitKey(1) == ord('q'):
            break

cap.release()
cv2.destroyAllWindows()
